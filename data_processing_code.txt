rules = { ‘gre_score’: (260, 340), ‘toefl_score’: (0, 120), ‘cgpa’:
(0.0, 4.0), ‘university_rating’: (1, 5), ‘sop’: (1.0, 5.0), ‘lor’: (1.0,
5.0), ‘research’: (0, 1), ‘chance_of_admit’: (0.0, 1.0) }

invalid_summary = {} invalid_indices = set()

for col, (low, high) in rules.items(): if col in df.columns: series =
pd.to_numeric(df[col], errors=‘coerce’) mask_invalid = (series < low) |
(series > high) | (series.isna()) count_invalid = mask_invalid.sum()
invalid_summary[col] = int(count_invalid)
invalid_indices.update(df[mask_invalid].index.tolist()) else:
invalid_summary[col] = ‘MISSING_COLUMN’

print(“counts per column (according to rules):”) for k, v in
invalid_summary.items(): print(f” - {k}: {v}“)

if invalid_indices: print(“invalid rows (first 20):”)
print(df.loc[sorted(list(invalid_indices))].head(20)) else:
print(“invalid rows found according to the rules.”)

report = pd.DataFrame([ {‘Check’: ‘initial_shape’, ‘Value’:
f”{df.shape}“}, {‘Check’: ‘missing_per_column’, ‘Value’:
df.isnull().sum().to_dict()}, {‘Check’: ‘duplicate_rows’, ‘Value’:
int(df.duplicated().sum())}, {‘Check’: ‘invalid_counts’, ‘Value’:
invalid_summary}, {‘Check’: ‘invalid_row_indices_sample’, ‘Value’:
list(sorted(list(invalid_indices)))[:20]}])

print(“report (summary):”) print(report.T)
